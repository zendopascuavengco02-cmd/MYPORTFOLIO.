130. Daily Sugar Consumption Tracker — Store daily sugar consumption for 10 days in a double[] array, accepting nonnegative values in grams. Calculate total and average consumption. Count days when consumption exceeded 50 grams. Display amounts in descending order.

Programming Language: Java

Subject: CC102

Project Type: Console-Based Application

Java Source Code

/*VENGCO, ZENDO PASCUA.
BSIT 1-C */
import java.util.Scanner;
import java.util.Arrays;

public class SugarTracker {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        // Store sugar consumption for 10 days
        double[] sugar = new double[10];

        double total = 0;
        double average;
        int count = 0;

        // Input sugar consumption for 10 days
        for (int i = 0; i < sugar.length; i++) {

            System.out.print("Enter sugar consumption for Day "
                    + (i + 1) + " in grams: ");

            sugar[i] = input.nextDouble();

            // Make sure the input is nonnegative
            while (sugar[i] < 0) {
                System.out.println("Invalid input! Enter a nonnegative value.");
                System.out.print("Enter sugar consumption for Day "
                        + (i + 1) + " in grams: ");

                sugar[i] = input.nextDouble();
            }

            // Calculate total consumption
            total += sugar[i];

            // Count days exceeding 50 grams
            if (sugar[i] > 50) {
                count++;
            }
        }

        // Calculate average consumption
        average = total / sugar.length;

        // Display total and average
        System.out.println("\n=== SUGAR CONSUMPTION REPORT ===");
        System.out.printf("Total consumption: %.2f grams%n", total);
        System.out.printf("Average consumption: %.2f grams%n", average);
        System.out.println("Days exceeding 50 grams: " + count);

        // Sort the array in ascending order
        Arrays.sort(sugar);

        // Display amounts in descending order
        System.out.println("\nSugar consumption in descending order:");

        for (int i = sugar.length - 1; i >= 0; i--) {
            System.out.printf("%.2f grams%n", sugar[i]);
        }

        input.close();
    }
}
